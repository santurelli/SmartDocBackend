			SELECT num_preventivo + 1 AS numero
			  FROM d_e_preventivi
			 WHERE num_preventivo + 1 NOT IN (SELECT num_preventivo 
			 								 FROM d_e_preventivi 
			 								WHERE fl_deleted = 0 
			 								AND EXTRACT(year FROM data_preventivo) = CASE WHEN ?::character varying IS NULL 
			 																	  THEN EXTRACT(year FROM CURRENT_DATE) 
			 																	  ELSE EXTRACT(year FROM TO_DATE(?, 'DD/MM/YYYY')) END
			 							   )
			   AND EXTRACT(year FROM data_preventivo) = CASE WHEN ?::character varying IS NULL 
			   											  THEN EXTRACT(year FROM CURRENT_DATE) 
			   											  ELSE EXTRACT(year FROM TO_DATE(?, 'DD/MM/YYYY')) END
			   AND fl_deleted = 0
		  GROUP BY numero
		  ORDER BY numero DESC
		     LIMIT 1
		</content>
	</query>
